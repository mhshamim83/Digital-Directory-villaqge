URL খালি রেখে সেভ করলে লোকাল মোডে ফেরত যাবে।

২. গুগল শিটে স্ক্রিপ্ট তৈরি করার নিয়ম (মাত্র ২ মিনিট):

  1. একটি নতুন Google Sheet খুলুন এবং প্রথম সারিতে হেডার দিন: Timestamp | Name | Designation | Phone | IsYouth | BloodGroup | IsGovt | GovtStatus | GovtDept | Address
  2. উপরের মেনু থেকে Extensions > Apps Script-এ যান।
  3. সেখানে থাকা সব কোড মুছে নিচের কোডটি পেস্ট করুন:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // নতুন রো হিসেবে ডেটা ইনসার্ট করা
    sheet.appendRow([
      new Date().toLocaleString('en-US', { timeZone: 'Asia/Dhaka' }),
      data.name,
      data.designation,
      data.phone,
      data.isYouth ? 'হ্যাঁ' : 'না',
      data.bloodGroup || '',
      data.isGovtEmployee ? 'হ্যাঁ' : 'না',
      data.govtStatus === 'retired' ? 'অবসরপ্রাপ্ত' : (data.isGovtEmployee ? 'কর্মরত' : ''),
      data.govtDept || '',
      data.address || ''
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success", 
      message: "Data saved successfully"
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error", 
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
  1. Apps Script এ ডান পাশের Deploy > New deployment বাটনে ক্লিক করুন।
  2. Select type হিসেবে Web app নির্বাচন করুন।
  3. Who has access অপশনটিতে অবশ্যই "Anyone" (যে কেউ) নির্বাচন করবেন।
  4. তারপর Deploy করে প্রাপ্ত Web app URL কপি করে উপরে বসিয়ে সেভ করুন!
